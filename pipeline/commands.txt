# Create a postgres container

docker run -it --rm \
  -e POSTGRES_USER='root' \
  -e POSTGRES_PASSWORD='root' \
  -e POSTGRES_DB='ny_taxi' \
  -v ny_taxi_postgres_data:/var/lib/postgresql \
  -p 5432:5432 \
  postgres:18

# Run script on local machine to postgres containter

uv run python ingest_data.py \
  taxi_ingest:v001 \
  --pg-user=root \
  --pg-pass=root \
  --pg-host=localhost \ 
  --pg-port=5432 \
  --pg-db=ny_taxi \

  --month=7 \

## Correct (best practice)
 
### Create a docker network first
docker network create pg-network

### Add that network to postgres container
docker run -it --rm \
  -e POSTGRES_USER='root' \
  -e POSTGRES_PASSWORD='root' \
  -e POSTGRES_DB='ny_taxi' \
  -v ny_taxi_postgres_data:/var/lib/postgresql \
  -p 5432:5432 \
  --network=pg-network \
  --name pgdatabase \
  postgres:18

### Add that network to taxi_ingest container
docker run -it --rm \
  --network=pg-network \
  taxi_ingest:v001 \
  --pg-user=root \
  --pg-pass=root \
  --pg-host=pgdatabase \
  --pg-port=5432 \
  --pg-db=ny_taxi \
  --target-table=yellow_taxi_trips \
  --year=2021 \
  --month=7 \
  --chunksize=100000
